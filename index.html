📖 Integration Examples

// Include the simple table creator module
<script src="simple-table-creator.js"></script>

// Create a sample employee table
SimpleTableCreator.createEmployeeTable()
    .then(tableName => {
        console.log('Created table:', tableName);
        alert('Employee table created successfully!');
    })
    .catch(error => {
        console.error('Error:', error);
        alert('Failed to create table: ' + error.message);
    });

// Create a sales data table
SimpleTableCreator.createSalesTable()
    .then(tableName => console.log('Sales table:', tableName));
// Create a custom table with your own data
const headers = ['Product', 'Price', 'Category', 'Stock'];
const data = [
    ['MacBook Pro', 2499.99, 'Electronics', 12],
    ['Office Chair', 299.99, 'Furniture', 25],
    ['Coffee Mug', 19.99, 'Office Supplies', 100]
];

SimpleTableCreator.createCustomTable('ProductCatalog', headers, data)
    .then(tableName => {
        console.log('Custom table created:', tableName);
        return SimpleTableCreator.formatTable(tableName, {
            style: 'TableStyleMedium2'
        });
    })
    .then(() => console.log('Table formatted successfully'))
    .catch(error => console.error('Error:', error));
// Apply custom formatting to a table
const formatOptions = {
    style: 'TableStyleDark1',
    headerColor: '#0078d4',
    headerFontColor: 'white'
};

// Get all table names first
SimpleTableCreator.getTableNames()
    .then(tableNames => {
        console.log('Available tables:', tableNames);
        
        // Format the first table
        if (tableNames.length > 0) {
            return SimpleTableCreator.formatTable(tableNames[0], formatOptions);
        }
    })
    .then(tableName => console.log('Formatted table:', tableName))
    .catch(error => console.error('Formatting error:', error));

🚀 Quick Start

To get started with these Excel table creators:

  1. For Office Add-ins: Use the manifest file and host these files on a web server
  2. For Excel Online: Load the HTML file directly in Excel Online's Script Lab
  3. For Testing: Use the Office Add-ins development tools and sideload the manifest
  4. For Integration: Include the JavaScript modules in your existing Office.js projects
